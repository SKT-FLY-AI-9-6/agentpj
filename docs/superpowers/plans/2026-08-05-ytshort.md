# ytshort 구현 계획

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 유튜브 외국어 영상의 하이라이트 구간을 잘라 한국어 자막과 나레이션을 얹은 세로형 숏츠를 만드는 Gradio 웹 데모를 Colab에서 동작시킨다.

**Architecture:** LangGraph 선형 파이프라인 8노드. 대본 단계에서 `interrupt()`로 멈추고 웹 UI가 대본을 편집·승인한다. 자연어 피드백이 있으면 supervisor가 되감을 노드를 정한다. 순수 함수(슬롯 계산, ass 생성, 상태 무효화)를 별도 모듈로 분리해 GPU 없이 테스트한다.

**Tech Stack:** Python 3.12, LangGraph, LangChain, Gradio, WhisperX, pyannote.audio, yt-dlp, ffmpeg, Typecast TTS, OpenAI, Tavily

## Global Constraints

- 스펙: `docs/superpowers/specs/2026-08-05-ytshort-design.md`

```python
_CLEARED: dict[str, dict] = {
    "highlight": {
        "clip_start": 0.0, "clip_end": 0.0, "clip_path": "", "audio_path": "",
        "segments": [], "segments_ko": [], "slots": [], "cues": [],
        "narration_audio": [], "dropped_narrations": [],
    },
    "translate": {
        "segments_ko": [], "slots": [], "cues": [],
        "narration_audio": [], "dropped_narrations": [],
    },
    "script": {
        "slots": [], "cues": [],
        "narration_audio": [], "dropped_narrations": [],
    },
    "tts": {
        "narration_audio": [], "dropped_narrations": [],
    },
    "FINISH": {},
}


def invalidate_downstream(state: dict, goto: str) -> dict:
    """되감을 노드보다 하류에 있는 필드를 비운 갱신 딕셔너리를 반환한다.

    입력 state는 변경하지 않는다. 반환값을 그대로 Command(update=...)에 넣는다.
    """
    if goto not in _CLEARED:
        raise ValueError(f"알 수 없는 되감기 대상입니다: {goto}")
    return {k: (list(v) if isinstance(v, list) else v)
            for k, v in _CLEARED[goto].items()}
```

`clip_path`와 `audio_path`가 빈 문자열이 되므로 테스트의 `in (None, [], 0.0)` 조건에 걸린다. 테스트를 `in (None, [], 0.0, "")`으로 고친다.

- [ ] **Step 3: `ytshort/slots.py` 구현**

```python
from ytshort.config import CHAR_SAFETY, MIN_GAP, SYLLABLES_PER_SEC
from ytshort.state import Segment, Slot


def estimate_max_chars(duration: float) -> int:
    """주어진 초 안에 읽을 수 있는 한국어 글자 수를 보수적으로 추정한다."""
    if duration <= 0:
        return 0
    return int(duration * SYLLABLES_PER_SEC * CHAR_SAFETY)


def compute_slots(segments: list[Segment], clip_duration: float) -> list[Slot]:
    """대사가 없는 구간 중 MIN_GAP 이상인 곳을 나레이션 슬롯으로 만든다.

    클립 앞뒤 여백도 슬롯이 된다. 세그먼트가 겹쳐 있으면 그 사이는 슬롯이 아니다.
    """
    ordered = sorted(segments, key=lambda s: s["start"])

    gaps: list[tuple[float, float]] = []
    cursor = 0.0
    for s in ordered:
        if s["start"] - cursor >= MIN_GAP:
            gaps.append((cursor, s["start"]))
        cursor = max(cursor, s["end"])
    if clip_duration - cursor >= MIN_GAP:
        gaps.append((cursor, clip_duration))

    slots: list[Slot] = []
    for i, (start, end) in enumerate(gaps):
        duration = round(end - start, 3)
        slots.append({
            "id": f"slot-{i}",
            "start": start,
            "end": end,
            "duration": duration,
            "max_chars": estimate_max_chars(duration),
        })
    return slots
```

```python
from ytshort.state import Cue

_HEADER = """[Script Info]
ScriptType: v4.00+
WrapStyle: 2
ScaledBorderAndShadow: yes
PlayResX: {width}
PlayResY: {height}

[V4+ Styles]
Format: Name, Fontname, Fontsize, PrimaryColour, OutlineColour, BackColour, Bold, Italic, BorderStyle, Outline, Shadow, Alignment, MarginL, MarginR, MarginV, Encoding

[Events]
Format: Layer, Start, End, Style, Name, MarginL, MarginR, MarginV, Effect, Text

def format_timecode(seconds: float) -> str:
    """ass의 H:MM:SS.cc 형식으로 변환한다. 센티초는 내림한다."""
    if seconds < 0:
        seconds = 0.0
    total_cs = int(seconds * 100)
    cs = total_cs % 100
    total_s = total_cs // 100
    s = total_s % 60
    m = (total_s // 60) % 60
    h = total_s // 3600
    return f"{h}:{m:02d}:{s:02d}.{cs:02d}"


def _escape(text: str) -> str:
    return (text.replace("\\", "\\\\")
                .replace("{", "(")
                .replace("}", ")")
                .replace("\r\n", "\\N")
                .replace("\n", "\\N"))


def cues_to_ass(cues: list[Cue], width: int = 1080, height: int = 1920) -> str:
    """큐 목록을 ass 자막 문자열로 만든다. kind가 스타일 이름이 된다."""
    out = [_HEADER.format(width=width, height=height, font=SUBTITLE_FONT)]
    for c in sorted(cues, key=lambda c: c["start"]):
        text = c["text"].strip()
        if not text:
            continue
        out.append(
            f"Dialogue: 0,{format_timecode(c['start'])},{format_timecode(c['end'])},"
            f"{c['kind']},,0,0,0,,{_escape(text)}"
        )
    return "\n".join(out)
```

```python
import os

from ytshort.config import WORK_DIR
from ytshort.errors import DownloadBlocked, UnsupportedVideo
from ytshort.state import ClipState

_BOT_MARKERS = ("sign in to confirm", "confirm you're not a bot")


def is_bot_block(message: str) -> bool:
    lowered = message.lower()
    return any(m in lowered for m in _BOT_MARKERS)


def parse_heatmap(info: dict) -> list[dict]:
    """yt-dlp의 heatmap을 {start, end, value} 목록으로 정규화한다."""
    raw = info.get("heatmap") or []
    return [
        {"start": h["start_time"], "end": h["end_time"], "value": h["value"]}
        for h in raw
    ]


def _fetch_info(url: str) -> tuple[dict, str]:
    """영상을 내려받고 (메타데이터, 파일경로)를 반환한다."""
    import yt_dlp

    os.makedirs(WORK_DIR, exist_ok=True)
    opts = {
        "format": "bestvideo[height<=1080]+bestaudio/best[height<=1080]",
        "outtmpl": os.path.join(WORK_DIR, "%(id)s.%(ext)s"),
        "merge_output_format": "mp4",
        "quiet": True,
        "no_warnings": True,
    }
    with yt_dlp.YoutubeDL(opts) as ydl:
        info = ydl.extract_info(url, download=True)
        path = ydl.prepare_filename(info)
    base, _ = os.path.splitext(path)
    return info, base + ".mp4"


def ingest_node(state: ClipState) -> dict:
    """영상과 메타데이터를 확보한다.

    업로드 파일이 주어지면 다운로드를 건너뛴다. 이 경우 heatmap이 없으므로
    clip_start와 clip_end가 반드시 함께 와야 한다.
    """
    local = state.get("local_video")
    if local:
        if state.get("clip_start") is None or state.get("clip_end") is None:
            raise UnsupportedVideo(
                "업로드한 영상은 시청 구간 정보가 없습니다. 사용할 구간의 시작과 끝 초를 입력해 주세요."
            )
        return {
            "source_path": local,
            "video_id": os.path.splitext(os.path.basename(local))[0],
            "title": os.path.basename(local),
            "source_duration": 0.0,
            "heatmap": [],
        }

    try:
        info, path = _fetch_info(state["url"])
    except Exception as exc:
        if is_bot_block(str(exc)):
            raise DownloadBlocked(
                "이 환경에서는 유튜브가 다운로드를 차단했습니다. "
                "영상 파일을 직접 올리고 사용할 구간을 입력해 주세요."
            ) from exc
        raise

    heatmap = parse_heatmap(info)
    if not heatmap:
        raise UnsupportedVideo(
            "이 영상에는 시청 구간 데이터가 없어 하이라이트를 자동으로 찾을 수 없습니다. "
            "영상 파일을 올리고 구간을 직접 지정하거나 다른 영상을 시도해 주세요."
        )

    return {
        "video_id": info["id"],
        "title": info.get("title", ""),
        "source_duration": float(info.get("duration") or 0.0),
        "source_path": path,
        "heatmap": heatmap,
    }
```

```python
import os
import subprocess

from ytshort.config import CLIP_DURATION, WORK_DIR
from ytshort.state import ClipState


def pick_window(
    heatmap: list[dict], source_duration: float, clip_duration: float
) -> tuple[float, float]:
    """가장 많이 시청된 지점을 중심으로 clip_duration 길이의 창을 잡는다.

    source_duration이 0이면 길이를 모른다는 뜻이므로 뒤쪽을 자르지 않는다.
    """
    if source_duration and source_duration <= clip_duration:
        return 0.0, source_duration

    peak = max(heatmap, key=lambda h: h["value"])
    centre = (peak["start"] + peak["end"]) / 2
    start = centre - clip_duration / 2

    if start < 0:
        start = 0.0
    if source_duration and start + clip_duration > source_duration:
        start = source_duration - clip_duration
    return round(start, 3), round(start + clip_duration, 3)


def _cut(source: str, start: float, end: float, video_id: str) -> tuple[str, str]:
    """클립 mp4와 16kHz mono wav를 만들고 두 경로를 반환한다."""
    os.makedirs(WORK_DIR, exist_ok=True)
    clip = os.path.join(WORK_DIR, f"{video_id}_clip.mp4")
    audio = os.path.join(WORK_DIR, f"{video_id}_clip.wav")
    duration = end - start

    subprocess.run(
        ["ffmpeg", "-y", "-ss", str(start), "-i", source, "-t", str(duration),
         "-c:v", "libx264", "-c:a", "aac", "-avoid_negative_ts", "make_zero", clip],
        check=True, capture_output=True,
    )
    subprocess.run(
        ["ffmpeg", "-y", "-i", clip, "-vn", "-ac", "1", "-ar", "16000", audio],
        check=True, capture_output=True,
    )
    return clip, audio


def highlight_node(state: ClipState) -> dict:
    """하이라이트 구간을 정하고 클립과 오디오를 추출한다."""
    if state.get("clip_start") is not None and state.get("clip_end"):
        start, end = float(state["clip_start"]), float(state["clip_end"])
    else:
        start, end = pick_window(
            state["heatmap"], state.get("source_duration", 0.0), CLIP_DURATION
        )

    clip, audio = _cut(state["source_path"], start, end, state["video_id"])
    return {"clip_start": start, "clip_end": end,
            "clip_path": clip, "audio_path": audio}
```

```python
from ytshort.config import (
    HF_TOKEN, WHISPER_COMPUTE_TYPE, WHISPER_DEVICE, WHISPER_MODEL,
)
from ytshort.errors import TranscriptionFailed
from ytshort.state import ClipState, Segment


def normalize(result: dict) -> list[Segment]:
    """WhisperX 결과를 Segment 목록으로 정규화한다.

    빈 텍스트 세그먼트를 버리고, 화자 정보가 없으면 UNKNOWN을 넣는다.
    타임코드는 이미 클립 기준이므로 그대로 쓴다.
    """
    segments: list[Segment] = []
    for s in result.get("segments", []):
        text = (s.get("text") or "").strip()
        if not text:
            continue
        segments.append({
            "start": float(s["start"]),
            "end": float(s["end"]),
            "speaker": s.get("speaker") or "UNKNOWN",
            "text": text,
        })
    return segments


def _run_whisperx(audio_path: str) -> dict:
    """전사, 강제정렬, 화자분리를 순서대로 수행한다.

    VRAM을 아끼려고 단계마다 모델을 내리고 다음 모델을 올린다.
    화자분리만 실패하면 예외를 삼키고 전사 결과를 그대로 반환한다.
    """
    import gc

    import torch
    import whisperx

    model = whisperx.load_model(
        WHISPER_MODEL, WHISPER_DEVICE, compute_type=WHISPER_COMPUTE_TYPE
    )
    audio = whisperx.load_audio(audio_path)
    result = model.transcribe(audio)
    language = result["language"]
    del model
    gc.collect()
    torch.cuda.empty_cache()

    align_model, meta = whisperx.load_align_model(
        language_code=language, device=WHISPER_DEVICE
    )
    result = whisperx.align(
        result["segments"], align_model, meta, audio, WHISPER_DEVICE
    )
    del align_model
    gc.collect()
    torch.cuda.empty_cache()

    try:
        diarize = whisperx.diarize.DiarizationPipeline(
            use_auth_token=HF_TOKEN, device=WHISPER_DEVICE
        )
        result = whisperx.assign_word_speakers(diarize(audio), result)
    except Exception:
        pass

    return result


def transcribe_node(state: ClipState) -> dict:
    """오디오를 전사하고 화자를 붙인다."""
    try:
        result = _run_whisperx(state["audio_path"])
    except Exception as exc:
        message = str(exc)
        if "out of memory" in message.lower():
            raise TranscriptionFailed(
                "GPU 메모리가 부족해 전사에 실패했습니다. "
                "WHISPER_MODEL을 medium이나 small로 낮춰 다시 시도해 주세요."
            ) from exc
        raise TranscriptionFailed(f"전사에 실패했습니다: {message}") from exc
```

```python
from pydantic import BaseModel, Field

from ytshort.config import DEFAULT_TONE, LLM_MODEL
from ytshort.state import ClipState, Segment

_BASE = """당신은 영상 자막을 한국어로 옮기는 번역가입니다.

번호가 붙은 대사가 주어집니다. 같은 개수의 번역문을 같은 순서로 돌려주세요.
대사를 합치거나 나누지 마세요. 한 줄도 빠뜨리지 마세요.
앞뒤 대사의 흐름을 보고 대명사와 생략된 주어를 자연스럽게 채우세요.

말투 지침:
{tone}"""

_DIRECTIVE = """

추가 지시:
{directive}"""


class Translation(BaseModel):
    lines: list[str] = Field(description="입력과 같은 순서, 같은 개수의 한국어 번역문")


def build_prompt(tone: str, directive: str | None) -> str:
    prompt = _BASE.format(tone=tone or DEFAULT_TONE)
    if directive:
        prompt += _DIRECTIVE.format(directive=directive)
    return prompt


def merge_translations(segments: list[Segment], texts: list[str]) -> list[Segment]:
    """번역문을 세그먼트에 합친다. 타임코드와 화자는 그대로 둔다."""
    if len(segments) != len(texts):
        raise ValueError(
            f"번역 결과 개수가 맞지 않습니다: 원본 {len(segments)}개, 번역 {len(texts)}개"
        )
    return [{**s, "text": t.strip()} for s, t in zip(segments, texts)]


def _translate(texts: list[str], prompt: str) -> list[str]:
    from langchain.chat_models import init_chat_model
    from langchain.messages import HumanMessage, SystemMessage

    model = init_chat_model(LLM_MODEL, temperature=0.3)
    numbered = "\n".join(f"{i}. {t}" for i, t in enumerate(texts))
    result = model.with_structured_output(Translation).invoke(
        [SystemMessage(content=prompt), HumanMessage(content=numbered)]
    )
    return result.lines


def translate_node(state: ClipState) -> dict:
    """대사를 한국어로 옮긴다. 타임코드는 건드리지 않는다."""
    segments = state["segments"]
    if not segments:
        return {"segments_ko": []}

    prompt = build_prompt(state.get("tone") or DEFAULT_TONE, state.get("directive"))
    texts = _translate([s["text"] for s in segments], prompt)
    return {"segments_ko": merge_translations(segments, texts)}
```

```python
import json

from pydantic import BaseModel, Field

from ytshort.config import DEFAULT_TONE, JOKE_DURATION, LLM_MODEL
from ytshort.slots import compute_slots
from ytshort.state import ClipState, Cue, Segment, Slot

_SYSTEM = """당신은 숏폼 영상의 대본을 쓰는 작가입니다.

주어진 영상 클립의 번역된 대사를 읽고, 두 가지를 만듭니다.

1. 나레이션 — 상황을 짚어주거나 맥락을 보태는 음성 해설.
   반드시 get_slots로 확인한 빈 구간 안에만 넣습니다.
   해당 슬롯의 max_chars를 절대 넘기지 마세요. 넘기면 잘려나갑니다.
   대사가 이미 설명하고 있는 내용을 되풀이하지 마세요.

2. 드립 — 화면에 잠깐 뜨는 자막. 음성으로 읽지 않으므로 길이 제약이 없습니다.
   짧고 치고 빠지게 씁니다. 20자를 넘기지 마세요.

영상에 나오는 인물, 밈, 사건을 모르겠으면 search_context로 찾아보세요.
한국 시청자가 모를 외국 맥락은 아는 것으로 바꿔서 설명해야 합니다.

나레이션은 슬롯당 최대 1개입니다. 드립은 클립 전체에서 최대 3개입니다.
넣을 만한 자리가 없으면 억지로 만들지 말고 적게 내놓으세요.

말투 지침:
{tone}"""

_DIRECTIVE = """

추가 지시:
{directive}"""


class NewCue(BaseModel):
    kind: str = Field(description="narration 또는 joke")
    start: float = Field(description="클립 시작 기준 초. narration은 슬롯 시작과 같게")
    text: str = Field(description="화면과 음성에 나갈 한국어 문장")
    slot_id: str | None = Field(default=None, description="narration이면 슬롯 id")


class ScriptDraft(BaseModel):
    cues: list[NewCue] = Field(description="새로 만든 나레이션과 드립")


def dialogue_cues(segments_ko: list[Segment]) -> list[Cue]:
    """번역된 대사를 그대로 dialogue 큐로 옮긴다."""
    return [
        {
            "id": f"d{i}",
            "kind": "dialogue",
            "start": s["start"],
            "end": s["end"],
            "speaker": s["speaker"],
            "text": s["text"],
            "slot_id": None,
        }
        for i, s in enumerate(segments_ko)
    ]


def clamp_cues(cues: list[Cue], slots: list[Slot], clip_duration: float) -> list[Cue]:
    """에이전트가 만든 큐를 물리적 제약에 맞춰 다듬는다.

    나레이션은 슬롯 경계에 스냅하고 max_chars로 자른다. 슬롯을 못 찾으면 버린다.
    드립은 시작 시각에서 JOKE_DURATION만큼만 띄우고 클립 끝에서 자른다.
    """
    by_id = {s["id"]: s for s in slots}
    out: list[Cue] = []

    for c in cues:
        if c["kind"] == "narration":
            slot = by_id.get(c["slot_id"] or "")
            if slot is None:
                continue
            out.append({**c,
                        "start": slot["start"],
                        "end": slot["end"],
                        "text": c["text"][: slot["max_chars"]]})
        elif c["kind"] == "joke":
            start = min(c["start"], clip_duration)
            out.append({**c,
                        "start": start,
                        "end": min(start + JOKE_DURATION, clip_duration)})
        else:
            out.append(c)

    return sorted(out, key=lambda c: c["start"])


def _invent(
    transcript: str, slots: list[Slot], tone: str, directive: str | None
) -> list[Cue]:
    """도구를 쥔 에이전트를 돌려 나레이션과 드립을 받아온다."""
    from langchain.agents import create_agent
    from langchain.chat_models import init_chat_model
    from langchain.messages import HumanMessage, SystemMessage
    from langchain.tools import tool
    from langchain_tavily import TavilySearch

    @tool
    def get_slots() -> str:
        """나레이션을 넣을 수 있는 빈 구간 목록. 시각, 길이, 최대 글자 수를 준다."""
        return json.dumps(slots, ensure_ascii=False)

    @tool
    def get_transcript() -> str:
        """화자와 시각이 붙은 번역 대사 전문."""
        return transcript

    search = TavilySearch(max_results=3)

    @tool
    def search_context(query: str) -> str:
        """영상에 나온 인물, 밈, 사건의 배경을 웹에서 찾는다."""
        return str(search.invoke({"query": query}))

    prompt = _SYSTEM.format(tone=tone)
    if directive:
        prompt += _DIRECTIVE.format(directive=directive)

    model = init_chat_model(LLM_MODEL, temperature=0.7)
    agent = create_agent(
        model=model,
        tools=[get_slots, get_transcript, search_context],
        system_prompt=prompt,
    )
    result = agent.invoke({
        "messages": [HumanMessage(content="이 클립의 나레이션과 드립을 만들어 주세요.")]
    })
    draft_text = result["messages"][-1].content

    draft = model.with_structured_output(ScriptDraft).invoke([
        SystemMessage(content="아래 작가의 결과물을 정해진 구조로 옮기세요. 내용을 바꾸지 마세요."),
        HumanMessage(content=draft_text),
    ])

    return [
        {
            "id": f"{c.kind[0]}{i}",
            "kind": c.kind,
            "start": c.start,
            "end": c.start,
            "speaker": None,
            "text": c.text,
            "slot_id": c.slot_id,
        }
        for i, c in enumerate(draft.cues)
        if c.kind in ("narration", "joke")
    ]


def script_node(state: ClipState) -> dict:
    """빈 슬롯을 계산하고 나레이션과 드립을 만들어 cues를 완성한다."""
    segments_ko = state["segments_ko"]
    clip_duration = state["clip_end"] - state["clip_start"]
    slots = compute_slots(segments_ko, clip_duration)

    transcript = "\n".join(
        f"[{s['start']:.1f}-{s['end']:.1f}] {s['speaker']}: {s['text']}"
        for s in segments_ko
    )
    invented = _invent(
        transcript, slots, state.get("tone") or DEFAULT_TONE, state.get("directive")
    )

    cues = clamp_cues(dialogue_cues(segments_ko) + invented, slots, clip_duration)
    return {"slots": slots, "cues": cues}
```

```python
from typing import Literal

from pydantic import BaseModel, Field

from ytshort.config import LLM_MODEL, MAX_REVISION
from ytshort.state import invalidate_downstream

_SYSTEM = """당신은 숏폼 제작 파이프라인의 감독입니다.
사용자가 완성된 대본을 보고 남긴 피드백을 읽고, 어느 단계부터 다시 해야 할지 정하세요.

- highlight : 잘라낸 구간 자체가 잘못 골라졌을 때
- translate : 번역의 말투나 정확도가 문제일 때
- script    : 나레이션이나 드립의 내용, 개수, 위치가 문제일 때
- tts       : 대본은 괜찮고 목소리나 읽는 속도만 문제일 때
- FINISH    : 되돌릴 필요가 없을 때

directive에는 해당 단계가 그대로 실행할 수 있는 구체적인 지시를 쓰세요.
"말투를 고쳐라" 같은 막연한 말이 아니라 무엇을 어떻게 바꿀지 적으세요.
되돌리기만 하면 같은 결과가 나옵니다."""


class Rewind(BaseModel):
    goto: Literal["highlight", "translate", "script", "tts", "FINISH"] = Field(
        description="다시 실행할 단계"
    )
    reason: str = Field(description="그렇게 정한 이유 한 문장")
    directive: str = Field(description="대상 단계에 전달할 구체적인 지시")


def _ask(feedback: str, cues_summary: str) -> Rewind:
    from langchain.chat_models import init_chat_model
    from langchain.messages import HumanMessage, SystemMessage

    model = init_chat_model(LLM_MODEL, temperature=0)
    return model.with_structured_output(Rewind).invoke([
        SystemMessage(content=_SYSTEM),
        HumanMessage(content=f"현재 대본:\n{cues_summary}\n\n사용자 피드백:\n{feedback}"),
    ])


def decide(feedback: str, cues_summary: str, revision: int) -> Rewind:
    """피드백을 읽고 되감을 단계를 정한다. 재시도 한도에 닿으면 무조건 진행한다."""
    if revision >= MAX_REVISION:
        return Rewind(
            goto="FINISH",
            reason=f"수정 횟수가 한도({MAX_REVISION})에 도달해 현재 대본으로 진행합니다.",
            directive="",
        )
    return _ask(feedback, cues_summary)


def apply_rewind(state: dict, rewind: Rewind) -> tuple[str, dict]:
    """되감기 결정을 (다음 노드, 상태 갱신)으로 바꾼다."""
    if rewind.goto == "FINISH":
        return "tts", {"directive": None, "feedback": None,
                       "revision": state.get("revision", 0)}

    update = invalidate_downstream(state, rewind.goto)
    update["directive"] = rewind.directive
    update["feedback"] = None
    update["revision"] = state.get("revision", 0) + 1
    return rewind.goto, update
```

```python
import time

import requests

from ytshort.config import TYPECAST_API_KEY, TYPECAST_TIMEOUT, TYPECAST_VOICE

_BASE = "https://typecast.ai/api"


def _headers() -> dict:
    if not TYPECAST_API_KEY:
        raise RuntimeError("TYPECAST_API_KEY가 설정되지 않았습니다.")
    return {"Authorization": f"Bearer {TYPECAST_API_KEY}"}


def request_speak(text: str) -> str:
    """합성을 요청하고 결과를 조회할 URL을 반환한다."""
    if not TYPECAST_VOICE:
        raise RuntimeError("TYPECAST_VOICE가 설정되지 않았습니다.")
    response = requests.post(
        f"{_BASE}/speak",
        headers=_headers(),
        json={
            "text": text,
            "lang": "auto",
            "actor_id": TYPECAST_VOICE,
            "xapi_hd": True,
            "model_version": "latest",
        },
        timeout=30,
    )
    response.raise_for_status()
    return response.json()["result"]["speak_v2_url"]


def poll_download_url(speak_url: str) -> str:
    """합성이 끝날 때까지 폴링하고 오디오 다운로드 URL을 반환한다."""
    deadline = time.time() + TYPECAST_TIMEOUT
    while time.time() < deadline:
        response = requests.get(speak_url, headers=_headers(), timeout=30)
        response.raise_for_status()
        result = response.json()["result"]
        if result["status"] == "done":
            return result["audio_download_url"]
        if result["status"] == "failed":
            raise RuntimeError("Typecast 합성이 실패했습니다.")
        time.sleep(1.0)
    raise TimeoutError("Typecast 합성이 제한 시간 안에 끝나지 않았습니다.")


def download(url: str, out_path: str) -> None:
    response = requests.get(url, timeout=60)
    response.raise_for_status()
    with open(out_path, "wb") as f:
        f.write(response.content)
```

```python
import os
import subprocess

from ytshort.config import LLM_MODEL, TTS_RETRY, WORK_DIR
from ytshort.state import ClipState
from ytshort.typecast import download, poll_download_url, request_speak


def audio_duration(path: str) -> float:
    """ffprobe로 오디오 길이(초)를 잰다."""
    result = subprocess.run(
        ["ffprobe", "-v", "error", "-show_entries", "format=duration",
         "-of", "default=noprint_wrappers=1:nokey=1", path],
        check=True, capture_output=True, text=True,
    )
    return float(result.stdout.strip())


def synthesize_to_file(text: str, out_path: str) -> float:
    """텍스트를 합성해 파일로 저장하고 실제 길이(초)를 반환한다."""
    os.makedirs(os.path.dirname(out_path) or ".", exist_ok=True)
    download(poll_download_url(request_speak(text)), out_path)
    return audio_duration(out_path)


def shorten(text: str, target_chars: int) -> str:
    """뜻을 유지한 채 문장을 target_chars 이하로 줄인다."""
    from langchain.chat_models import init_chat_model
    from langchain.messages import HumanMessage, SystemMessage

    model = init_chat_model(LLM_MODEL, temperature=0.3)
    result = model.invoke([
        SystemMessage(content=(
            f"주어진 한국어 문장을 {target_chars}자 이하로 줄이세요. "
            "핵심 뜻과 말투를 유지하고, 문장이 끊긴 느낌이 들지 않게 하세요. "
            "줄인 문장만 출력하세요."
        )),
        HumanMessage(content=text),
    ])
    return result.content.strip()[:target_chars]


def tts_node(state: ClipState) -> dict:
    """나레이션 큐만 합성한다. 슬롯을 넘기면 줄여 재합성하고, 안 되면 버린다."""
    slots = {s["id"]: s for s in state.get("slots", [])}
    kept: list[dict] = []
    dropped: list[dict] = []

    for cue in state.get("cues", []):
        if cue["kind"] != "narration":
            continue

        slot = slots.get(cue["slot_id"] or "")
        if slot is None:
            dropped.append({"cue_id": cue["id"], "text": cue["text"],
                            "reason": "해당하는 빈 구간을 찾지 못했습니다."})
            continue

        text = cue["text"]
        path = os.path.join(WORK_DIR, f"narr_{cue['id']}.wav")
        placed = False

        for attempt in range(TTS_RETRY + 1):
            try:
                duration = synthesize_to_file(text, path)
            except Exception as exc:
                dropped.append({"cue_id": cue["id"], "text": text,
                                "reason": f"음성 합성에 실패했습니다: {exc}"})
                break

            if duration <= slot["duration"]:
                kept.append({"cue_id": cue["id"], "path": path,
                             "duration": duration, "start": slot["start"]})
                placed = True
                break

            if attempt == TTS_RETRY:
                break

            ratio = slot["duration"] / duration
            text = shorten(text, max(1, int(len(text) * ratio * 0.9)))

        if not placed and not any(d["cue_id"] == cue["id"] for d in dropped):
            dropped.append({"cue_id": cue["id"], "text": text,
                            "reason": "빈 구간 길이를 맞추지 못했습니다."})

    return {"narration_audio": kept, "dropped_narrations": dropped}
```

```python
import os
import subprocess

from ytshort.config import SUBTITLE_FONT, WORK_DIR
from ytshort.state import ClipState

WIDTH, HEIGHT = 1080, 1920


def build_filter(narrations: list[dict]) -> tuple[str, list[str]]:
    """ffmpeg 필터 그래프와 추가 입력 파일 목록을 만든다.

    영상은 세로 캔버스에 맞추고 배경은 원본을 확대·블러 처리해 채운다.
    나레이션은 각자의 시작 시각에 지연시켜 원본 오디오와 섞는다.
    """
    video = (
        f"[0:v]scale={WIDTH}:{HEIGHT}:force_original_aspect_ratio=increase,"
        f"crop={WIDTH}:{HEIGHT},boxblur=40:5[bg];"
        f"[0:v]scale={WIDTH}:-2[fg];"
        f"[bg][fg]overlay=(W-w)/2:(H-h)/2[v]"
    )

    if not narrations:
        return video, []

    parts = [video]
    labels = ["[0:a]"]
    for i, n in enumerate(narrations):
        delay_ms = int(n["start"] * 1000)
        parts.append(f"[{i + 1}:a]adelay={delay_ms}|{delay_ms}[n{i}]")
        labels.append(f"[n{i}]")

    parts.append(
        "".join(labels) + f"amix=inputs={len(labels)}:normalize=0:dropout_transition=0[a]"
    )
    return ";".join(parts), [n["path"] for n in narrations]


def render_node(state: ClipState) -> dict:
    """크롭, 오디오 믹싱, 자막 번인을 한 번에 처리한다."""
    os.makedirs(WORK_DIR, exist_ok=True)
    out_path = os.path.join(WORK_DIR, f"{state['video_id']}_short.mp4")
    narrations = state.get("narration_audio", [])
    graph, extra_inputs = build_filter(narrations)

    ass = state["ass_path"].replace("\\", "/").replace(":", "\\:")
    graph += f";[v]subtitles='{ass}':fontsdir=/usr/share/fonts[vout]"

    cmd = ["ffmpeg", "-y", "-i", state["clip_path"]]
    for path in extra_inputs:
        cmd += ["-i", path]
    cmd += ["-filter_complex", graph, "-map", "[vout]"]
    cmd += ["-map", "[a]" if narrations else "0:a"]
    cmd += ["-c:v", "libx264", "-preset", "veryfast", "-crf", "23",
            "-c:a", "aac", "-shortest", out_path]

    subprocess.run(cmd, check=True, capture_output=True)
    return {"output_path": out_path}
```

`SUBTITLE_FONT`는 ass 파일이 스타일에 이름을 적어두므로 여기서는 `fontsdir`만 지정한다.

```python
from typing import Literal

from langgraph.graph import END, START, StateGraph
from langgraph.types import Command, interrupt

from ytshort.nodes import render, subtitle
from ytshort.nodes.highlight import highlight_node
from ytshort.nodes.ingest import ingest_node
from ytshort.nodes.script import script_node
from ytshort.nodes.transcribe import transcribe_node
from ytshort.nodes.translate import translate_node
from ytshort.nodes.tts import tts_node
from ytshort.state import ClipState
from ytshort.supervisor import apply_rewind, decide


def summarize_cues(cues: list[dict]) -> str:
    """supervisor에게 넘길 대본 요약."""
    return "\n".join(
        f"[{c['start']:.1f}s] ({c['kind']}) {c['text']}" for c in cues
    )


def review_node(
    state: ClipState,
) -> Command[Literal["highlight", "translate", "script", "tts"]]:
    """대본을 사람에게 보여주고 응답에 따라 진행하거나 되감는다."""
    answer = interrupt({
        "cues": state["cues"],
        "clip_path": state.get("clip_path", ""),
        "dropped_narrations": state.get("dropped_narrations", []),
    })

    cues = answer.get("cues") or state["cues"]
    feedback = (answer.get("feedback") or "").strip()

    if not feedback:
        return Command(goto="tts", update={"cues": cues, "directive": None})

    rewind = decide(feedback, summarize_cues(cues), state.get("revision", 0))
    goto, update = apply_rewind({**state, "cues": cues}, rewind)
    update.setdefault("cues", cues)
    return Command(goto=goto, update=update)


def subtitle_node(state: ClipState) -> dict:
    return subtitle.subtitle_node(state)


def render_node(state: ClipState) -> dict:
    return render.render_node(state)


def build_graph(checkpointer=None):
    """파이프라인 그래프를 조립한다. checkpointer는 interrupt 재개에 필요하다."""
    builder = StateGraph(ClipState)

    builder.add_node("ingest", ingest_node)
    builder.add_node("highlight", highlight_node)
    builder.add_node("transcribe", transcribe_node)
    builder.add_node("translate", translate_node)
    builder.add_node("script", script_node)
    builder.add_node("review", review_node)
    builder.add_node("tts", tts_node)
    builder.add_node("subtitle", subtitle_node)
    builder.add_node("render", render_node)

    builder.add_edge(START, "ingest")
    builder.add_edge("ingest", "highlight")
    builder.add_edge("highlight", "transcribe")
    builder.add_edge("transcribe", "translate")
    builder.add_edge("translate", "script")
    builder.add_edge("script", "review")
    builder.add_edge("tts", "subtitle")
    builder.add_edge("subtitle", "render")
    builder.add_edge("render", END)

    return builder.compile(checkpointer=checkpointer)
```

`review_node`가 `Command(goto=...)`로 되감으므로 `review`에서 나가는 정적 엣지는 만들지 않는다. `highlight`로 되감으면 그 뒤 정적 엣지를 따라 다시 `review`에 도달한다.

```python
import gradio as gr
from langgraph.checkpoint.sqlite import SqliteSaver
from langgraph.types import Command

from ytshort.errors import YtshortError
from ytshort.graph import build_graph

CHECKPOINTER = SqliteSaver(
    sqlite3.connect(os.path.join(WORK_DIR, "checkpoints.db"), check_same_thread=False)
)
GRAPH = build_graph(checkpointer=CHECKPOINTER)

STEP_LABELS = {
    "ingest": "영상을 받는 중",
    "highlight": "하이라이트 구간을 자르는 중",
    "transcribe": "음성을 받아쓰고 화자를 나누는 중",
    "translate": "한국어로 옮기는 중",
    "script": "대본을 쓰는 중",
    "review": "대본을 검토 대기 중",
    "tts": "나레이션을 만드는 중",
    "subtitle": "자막을 만드는 중",
    "render": "영상을 합치는 중",
}

    "자연스러운 대화체 (기본)": DEFAULT_TONE,
    "차분한 다큐 나레이션": "차분하고 담담한 해설체로 옮길 것. 존댓말을 쓰고 과장을 피한다.",
    "가볍고 빠른 리액션": "짧고 빠르게 치고 빠지는 반말. 감탄사와 리액션을 살린다.",
}


def cues_to_rows(cues):
    return [[c["id"], round(c["start"], 1), round(c["end"], 1),
             c["kind"], c["speaker"] or "", c["text"]] for c in cues]


def rows_to_cues(rows, original):
    by_id = {c["id"]: c for c in original}
    out = []
    for row in rows:
        cue_id = str(row[0])
        base = by_id.get(cue_id)
        if base is None:
            continue
        out.append({**base, "text": str(row[5])})
    return out


def _interrupt_payload(config):
    """현재 스레드가 멈춰 있으면 interrupt 값을, 아니면 None을 반환한다."""
    tasks = GRAPH.get_state(config).tasks
    for task in tasks:
        if task.interrupts:
            return task.interrupts[0].value
    return None


def _drive(stream_input, config, log):
    """그래프를 돌리며 진행 로그를 yield하고, 멈추면 대본 화면을 연다."""
    try:
        for chunk in GRAPH.stream(stream_input, config=config, stream_mode="updates"):
            for node in chunk:
                if node == "__interrupt__":
                    continue
                log.append(f"• {STEP_LABELS.get(node, node)} 완료")
                yield "\n".join(log), gr.update(), gr.update(), gr.update(), gr.update()
    except YtshortError as exc:
        log.append(f"\n중단되었습니다.\n{exc}")
        yield "\n".join(log), gr.update(), gr.update(), gr.update(), gr.update()
        return
    except Exception as exc:
        traceback.print_exc()
        log.append(f"\n예상치 못한 오류입니다: {exc}")
        yield "\n".join(log), gr.update(), gr.update(), gr.update(), gr.update()
        return

    payload = _interrupt_payload(config)
    if payload is not None:
        log.append("\n대본이 준비됐습니다. 아래에서 확인해 주세요.")
        yield (
            "\n".join(log),
            gr.update(visible=True),
            cues_to_rows(payload["cues"]),
            gr.update(value=payload.get("clip_path") or None),
            gr.update(visible=False),
        )
        return

    final = GRAPH.get_state(config).values
    dropped = final.get("dropped_narrations", [])
    if dropped:
        log.append("\n다음 나레이션은 넣지 못했습니다.")
        for d in dropped:
            log.append(f"  - {d['text']} ({d['reason']})")
    log.append("\n완성됐습니다.")
    yield (
        "\n".join(log),
        gr.update(visible=False),
        gr.update(),
        gr.update(),
        gr.update(value=final.get("output_path"), visible=True),
    )


def start(url, uploaded, tone_key, tone_custom, clip_start, clip_end, thread_id):
    config = {"configurable": {"thread_id": thread_id}}
    tone = tone_custom.strip() or TONE_PRESETS[tone_key]

    payload = {"tone": tone, "revision": 0, "url": url.strip(), "local_video": None}
    if uploaded:
        payload["local_video"] = uploaded
        payload["clip_start"] = float(clip_start)
        payload["clip_end"] = float(clip_end)
    elif not url.strip():
        yield "유튜브 주소를 넣거나 영상 파일을 올려 주세요.", gr.update(), gr.update(), gr.update(), gr.update()
        return

    yield from _drive(payload, config, ["시작합니다."])


def resume(rows, feedback, table_state, thread_id):
    config = {"configurable": {"thread_id": thread_id}}
    payload = _interrupt_payload(config)
    if payload is None:
        yield "진행 중인 작업이 없습니다. 처음부터 다시 시작해 주세요.", gr.update(), gr.update(), gr.update(), gr.update()
        return

    cues = rows_to_cues(rows, payload["cues"])
    yield from _drive(
        Command(resume={"cues": cues, "feedback": feedback}),
        config,
        ["반영하는 중입니다."],
    )


with gr.Blocks(title="ytshort") as demo:
    thread_id = gr.State(lambda: str(uuid.uuid4()))
    table_state = gr.State([])

    gr.Markdown("## 유튜브 영상을 한국어 숏츠로")

    with gr.Row():
        url_in = gr.Textbox(label="유튜브 주소", placeholder="https://youtu.be/...", scale=3)
        upload_in = gr.File(label="또는 영상 파일 올리기", type="filepath", scale=2)

    with gr.Row():
        clip_start_in = gr.Number(label="시작 초 (파일을 올린 경우)", value=0)
        clip_end_in = gr.Number(label="끝 초 (파일을 올린 경우)", value=25)

    tone_in = gr.Dropdown(
        label="말투", choices=list(TONE_PRESETS), value=list(TONE_PRESETS)[0]
    )
    tone_custom_in = gr.Textbox(
        label="말투를 직접 쓰고 싶다면", placeholder="비워두면 위 선택을 씁니다."
    )
    start_btn = gr.Button("만들기", variant="primary")

    log_out = gr.Textbox(label="진행 상황", lines=10, interactive=False)

    with gr.Group(visible=False) as review_group:
        gr.Markdown("### 대본을 확인해 주세요")
        clip_preview = gr.Video(label="자막 없는 원본 구간")
        cue_table = gr.Dataframe(
            headers=["id", "시작", "끝", "종류", "화자", "내용"],
            datatype=["str", "number", "number", "str", "str", "str"],
            label="내용 칸만 고치면 됩니다",
            interactive=True,
        )
        feedback_in = gr.Textbox(
            label="다시 만들고 싶은 부분이 있다면 적어주세요",
            placeholder="비워두면 위 표 그대로 진행합니다.",
        )
        resume_btn = gr.Button("이대로 진행", variant="primary")

    result_out = gr.Video(label="완성된 숏츠", visible=False)

    outputs = [log_out, review_group, cue_table, clip_preview, result_out]
    start_btn.click(

        [url_in, upload_in, tone_in, tone_custom_in, clip_start_in, clip_end_in, thread_id],

demo.queue(default_concurrency_limit=1)

    main()
```
